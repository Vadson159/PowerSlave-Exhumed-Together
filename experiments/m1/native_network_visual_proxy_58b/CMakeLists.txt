cmake_minimum_required(VERSION 3.24)
project(pse_native_network_visual_proxy_58b LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
# Reuse the frozen DOC56 chain (DOC55 -> DOC53 -> DOC51 -> network_input/ENet) unchanged.
add_subdirectory(../native_map_ready integration)
# DOC58B experimental client: a derived copy of the DOC56 client, never replacing it.
add_library(pse_native_client58b SHARED src/client58b.cpp)
target_compile_definitions(pse_native_client58b PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX PSE_DOC55_SNAPSHOT PSE_DOC56_MAPREADY PSE_DOC58B)
target_include_directories(pse_native_client58b PRIVATE src)
target_compile_options(pse_native_client58b PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(pse_native_client58b PRIVATE m12a_minhook pse_core bcrypt)
target_link_options(pse_native_client58b PRIVATE /INCREMENTAL:NO /DEBUG:FULL /OPT:REF /OPT:ICF)
add_executable(proxy58b_contract tests/proxy58b_contract.cpp)
target_include_directories(proxy58b_contract PRIVATE src)
target_link_libraries(proxy58b_contract PRIVATE pse_core)
target_compile_definitions(proxy58b_contract PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(proxy58b_contract PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
