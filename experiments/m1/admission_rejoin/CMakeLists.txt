cmake_minimum_required(VERSION 3.24)
project(pse_admission_rejoin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/enet-prototype" transport EXCLUDE_FROM_ALL)
add_library(admission_session STATIC ../network_input/src/network_input.cpp)
target_include_directories(admission_session PUBLIC ../network_input/src)
target_compile_definitions(admission_session PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(admission_session PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(admission_session PUBLIC pse_core)
add_executable(r2_original tests/r2_original.cpp)
target_link_libraries(r2_original PRIVATE admission_session)
add_executable(r2_fixed tests/r2_fixed.cpp)
target_link_libraries(r2_fixed PRIVATE admission_session)
add_executable(admission_contract tests/admission_contract.cpp)
target_compile_options(admission_contract PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(admission_contract PRIVATE admission_session)
add_executable(r1_live ../generation_routing/tests/r1_live.cpp)
target_link_libraries(r1_live PRIVATE admission_session)
add_executable(routing_contract ../generation_routing/tests/routing_contract.cpp)
target_link_libraries(routing_contract PRIVATE admission_session)
