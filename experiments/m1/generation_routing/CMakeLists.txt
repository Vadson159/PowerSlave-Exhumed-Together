cmake_minimum_required(VERSION 3.24)
project(pse_generation_routing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/enet-prototype" transport EXCLUDE_FROM_ALL)
set(SESSION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../network_input/src")
add_library(routing_session STATIC "${SESSION_DIR}/network_input.cpp")
target_include_directories(routing_session PUBLIC "${SESSION_DIR}")
target_compile_definitions(routing_session PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(routing_session PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(routing_session PUBLIC pse_core)
foreach(name IN ITEMS r1_original r1_live)
  add_executable(${name} tests/${name}.cpp)
  target_compile_options(${name} PRIVATE /EHsc /utf-8)
  target_link_libraries(${name} PRIVATE routing_session)
endforeach()
add_executable(routing_contract tests/routing_contract.cpp)
target_compile_options(routing_contract PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(routing_contract PRIVATE routing_session)
# Compile the unchanged native consumer against the amended internal DTOs.
# No second DLL, linker injection, native load or gameplay experiment.
add_library(native_compile_check OBJECT ../native_bootstrap/src/bootstrap.cpp)
target_compile_definitions(native_compile_check PRIVATE PSE_M12A UNICODE _UNICODE)
target_compile_options(native_compile_check PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_include_directories(native_compile_check PRIVATE ../native_bootstrap/third_party/minhook/include)
target_link_libraries(native_compile_check PRIVATE routing_session)
