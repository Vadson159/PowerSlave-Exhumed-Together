cmake_minimum_required(VERSION 3.24)
project(pse_cadence49 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(TRANSPORT "${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/enet-prototype")
add_subdirectory(../network_input integration)
add_library(qa_enet STATIC
  ${TRANSPORT}/third_party/enet/callbacks.c ${TRANSPORT}/third_party/enet/compress.c
  ${TRANSPORT}/third_party/enet/host.c ${TRANSPORT}/third_party/enet/list.c
  ${TRANSPORT}/third_party/enet/packet.c qa_enet/peer.c qa_enet/protocol.c qa_enet/win32.c)
target_include_directories(qa_enet PUBLIC ${TRANSPORT}/third_party/enet/include)
target_link_libraries(qa_enet PUBLIC ws2_32 winmm)
get_target_property(core_sources pse_core SOURCES)
get_target_property(core_dir pse_core SOURCE_DIR)
list(TRANSFORM core_sources PREPEND "${core_dir}/")
add_library(qa_core STATIC ${core_sources})
target_include_directories(qa_core PUBLIC ${TRANSPORT}/include ${TRANSPORT}/third_party/enet/include)
target_compile_definitions(qa_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
target_link_libraries(qa_core PUBLIC qa_enet ws2_32 winmm psapi bcrypt)
add_library(qa_session STATIC ../network_input/src/network_input.cpp)
target_include_directories(qa_session PUBLIC ../network_input/src)
target_link_libraries(qa_session PUBLIC qa_core)
add_executable(before_qa_client before_native_client.cpp telemetry.cpp)
target_link_libraries(before_qa_client PRIVATE qa_core)
add_executable(after_qa_client ../network_input/src/native_client.cpp telemetry.cpp)
target_link_libraries(after_qa_client PRIVATE qa_core)
add_executable(cadence_host cadence_host.cpp telemetry.cpp)
target_link_libraries(cadence_host PRIVATE qa_session)
add_executable(timeout_contract timeout_contract.cpp)
target_link_libraries(timeout_contract PRIVATE m12a_adapter)
add_executable(r1_live ../generation_routing/tests/r1_live.cpp)
add_executable(r2_fixed ../admission_rejoin/tests/r2_fixed.cpp)
add_executable(routing_contract ../generation_routing/tests/routing_contract.cpp)
add_executable(admission_contract ../admission_rejoin/tests/admission_contract.cpp)
foreach(t r1_live r2_fixed routing_contract admission_contract)
  target_link_libraries(${t} PRIVATE m12a_adapter)
endforeach()
