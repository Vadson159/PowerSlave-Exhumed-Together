cmake_minimum_required(VERSION 3.24)
project(pse_m12a LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../native_bootstrap")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/enet-prototype" transport EXCLUDE_FROM_ALL)
add_library(m12a_minhook STATIC
  ${NATIVE}/third_party/minhook/src/buffer.c
  ${NATIVE}/third_party/minhook/src/hook.c
  ${NATIVE}/third_party/minhook/src/trampoline.c
  ${NATIVE}/third_party/minhook/src/hde/hde64.c)
target_include_directories(m12a_minhook PUBLIC ${NATIVE}/third_party/minhook/include)
add_library(m12a_adapter STATIC src/network_input.cpp)
target_include_directories(m12a_adapter PUBLIC src)
target_link_libraries(m12a_adapter PUBLIC pse_core)
target_compile_features(m12a_adapter PUBLIC cxx_std_20)
target_compile_options(m12a_adapter PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_compile_definitions(m12a_adapter PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(pse_m12a SHARED ${NATIVE}/src/bootstrap.cpp)
target_compile_features(pse_m12a PRIVATE cxx_std_20)
target_compile_definitions(pse_m12a PRIVATE PSE_M12A UNICODE _UNICODE)
target_compile_options(pse_m12a PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(pse_m12a PRIVATE m12a_minhook m12a_adapter bcrypt)
target_link_options(pse_m12a PRIVATE /INCREMENTAL:NO /DEBUG:FULL /OPT:REF /OPT:ICF)
# One loaded game DLL, retaining the accepted loader/export surface.
set_target_properties(pse_m12a PROPERTIES OUTPUT_NAME pse_m11_bootstrap)
add_executable(m12a_generation_test tests/generation_test.cpp)
target_link_libraries(m12a_generation_test PRIVATE pse_core)
add_executable(m12a_adapter_test tests/adapter_test.cpp)
target_link_libraries(m12a_adapter_test PRIVATE m12a_adapter)
add_executable(m12a_pressure_test tests/pressure_test.cpp)
target_link_libraries(m12a_pressure_test PRIVATE m12a_adapter)
add_executable(m12a_native_client src/native_client.cpp)
target_link_libraries(m12a_native_client PRIVATE pse_core)
