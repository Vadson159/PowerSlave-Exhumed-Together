cmake_minimum_required(VERSION 3.24)
project(pse_native_snapshot LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_subdirectory(../background_host integration)
target_compile_definitions(pse_m12a PRIVATE PSE_DOC55_SNAPSHOT)
target_compile_definitions(pse_native_client PRIVATE PSE_DOC55_SNAPSHOT)
add_executable(snapshot_contract tests/snapshot_contract.cpp)
target_link_libraries(snapshot_contract PRIVATE m12a_adapter)
target_compile_options(snapshot_contract PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
