cmake_minimum_required(VERSION 3.24)
project(pse_native_map_ready LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_subdirectory(../native_snapshot integration)
target_compile_definitions(pse_m12a PRIVATE PSE_DOC56_MAPREADY)
target_compile_definitions(pse_native_client PRIVATE PSE_DOC56_MAPREADY)
add_executable(map_ready_contract tests/map_ready_contract.cpp)
target_link_libraries(map_ready_contract PRIVATE m12a_adapter bcrypt)
target_compile_options(map_ready_contract PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
