cmake_minimum_required(VERSION 3.24)
project(pse_window_close_48 LANGUAGES C CXX)
# Build current sources into a fresh tree; never overwrite accepted DOC-44 DLLs.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_subdirectory(../network_input integration)
add_executable(r1_live ../generation_routing/tests/r1_live.cpp)
target_link_libraries(r1_live PRIVATE m12a_adapter)
add_executable(routing_contract ../generation_routing/tests/routing_contract.cpp)
target_link_libraries(routing_contract PRIVATE m12a_adapter)
add_executable(r2_fixed ../admission_rejoin/tests/r2_fixed.cpp)
target_link_libraries(r2_fixed PRIVATE m12a_adapter)
add_executable(admission_contract ../admission_rejoin/tests/admission_contract.cpp)
target_link_libraries(admission_contract PRIVATE m12a_adapter)
foreach(target r1_live routing_contract r2_fixed admission_contract)
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endforeach()
