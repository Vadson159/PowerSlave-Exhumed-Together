cmake_minimum_required(VERSION 3.24)
project(pse_sequence50 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_subdirectory(../admission_rejoin integration)
add_executable(sequence_contract sequence_contract.cpp)
target_link_libraries(sequence_contract PRIVATE admission_session)
target_compile_options(sequence_contract PRIVATE /W4 /WX /EHsc /utf-8)
add_executable(sequence_live sequence_live.cpp)
target_link_libraries(sequence_live PRIVATE admission_session)
target_compile_options(sequence_live PRIVATE /W4 /WX /EHsc /utf-8)
add_executable(timeout_contract ../input_cadence/timeout_contract.cpp)
target_link_libraries(timeout_contract PRIVATE admission_session)
add_library(native_compile_check OBJECT ../native_bootstrap/src/bootstrap.cpp)
target_compile_definitions(native_compile_check PRIVATE PSE_M12A UNICODE _UNICODE)
target_include_directories(native_compile_check PRIVATE ../native_bootstrap/third_party/minhook/include)
target_compile_options(native_compile_check PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(native_compile_check PRIVATE admission_session)
