cmake_minimum_required(VERSION 3.24)
project(pse_native_second_instance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_subdirectory(../network_input integration)
add_library(pse_native_client SHARED src/client.cpp)
target_compile_definitions(pse_native_client PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(pse_native_client PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(pse_native_client PRIVATE m12a_minhook pse_core bcrypt)
target_link_options(pse_native_client PRIVATE /INCREMENTAL:NO /DEBUG:FULL /OPT:REF /OPT:ICF)
foreach(pair "r1_live;generation_routing/tests/r1_live.cpp" "routing_contract;generation_routing/tests/routing_contract.cpp" "r2_fixed;admission_rejoin/tests/r2_fixed.cpp" "admission_contract;admission_rejoin/tests/admission_contract.cpp" "timeout_contract;input_cadence/timeout_contract.cpp" "sequence_contract;input_sequence/sequence_contract.cpp" "sequence_live;input_sequence/sequence_live.cpp")
  list(GET pair 0 name)
  list(GET pair 1 source)
  add_executable(${name} ../${source})
  target_link_libraries(${name} PRIVATE m12a_adapter)
endforeach()
add_executable(native_client_contract tests/client_worker_test.cpp)
target_link_libraries(native_client_contract PRIVATE pse_core)
target_compile_options(native_client_contract PRIVATE /W4 /WX /EHsc /utf-8)
