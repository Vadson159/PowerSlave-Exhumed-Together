cmake_minimum_required(VERSION 3.20)
project(PSE_P25_Diagnostics LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pse-diagnostics supervisor.cpp)
target_compile_features(pse-diagnostics PRIVATE cxx_std_17)
target_compile_definitions(pse-diagnostics PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(pse-diagnostics PRIVATE /W4 /utf-8 /Zi)
target_link_options(pse-diagnostics PRIVATE /DEBUG)
target_link_libraries(pse-diagnostics PRIVATE dbghelp)
# Fixture stays in the build tree. It is never shipped to friends.
add_executable(p25-crash-fixture crash_fixture.cpp)
enable_testing()
add_test(NAME diagnostics_help COMMAND pse-diagnostics --help)
set_tests_properties(diagnostics_help PROPERTIES PASS_REGULAR_EXPRESSION "standalone ENet only")
